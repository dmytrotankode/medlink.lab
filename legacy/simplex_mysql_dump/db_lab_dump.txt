==================== PROCEDURE parse_lab_message ====================
CREATE DEFINER=`root`@`localhost` PROCEDURE `parse_lab_message`(
  IN p_analyzer_message mediumtext, 
  IN p_analyzer_id mediumint, 
  IN p_mode varchar(64),
  OUT p_barcode varchar(512),
  OUT p_result_code varchar(64),
  OUT p_result_message mediumtext
  )
    COMMENT 'v.2.1.182'
BEGIN
    DECLARE f  varchar(32);
    DECLARE t_str varchar(3024);
    DECLARE protocol_type varchar(32);

    INSERT INTO log_exchange(is_ok, log_exchange, log_mode) VALUES(0, p_analyzer_message, CONCAT('ACONNECT', p_analyzer_id));

    SELECT aat.analyzer_type_code, aa.analyzer_name, ifnull(JSON_UNQUOTE(JSON_EXTRACT(aa.analyzer_config, '$.protocol_type')),'') 
    INTO @analyzer_type_code, @analyzer_name, protocol_type
    FROM ac_analyzer aa JOIN ac_analyzer_type aat ON aa.analyzer_type_id = aat.analyzer_type_id WHERE aa.analyzer_id = p_analyzer_id;

    IF @analyzer_type_code not IN ('IRISIQ200', 'RAPID', 'Phadia 250') THEN
      WHILE LOCATE(CHAR(23), p_analyzer_message) > 0 DO 
        set p_analyzer_message = REPLACE(p_analyzer_message, SUBSTR(p_analyzer_message, LOCATE(CHAR(23), p_analyzer_message), 4), '');
      END WHILE;
  
      WHILE LOCATE(CHAR(2), p_analyzer_message) > 0 DO --                                                              біло 2|
        set p_analyzer_message = REPLACE(p_analyzer_message, SUBSTR(p_analyzer_message, LOCATE(CHAR(2), p_analyzer_message), 2), '');
      END WHILE;
  
      set p_analyzer_message = REPLACE(p_analyzer_message, char(3), '');
    END IF;

    SET @qrq_message = '';

    set @date = DATE_FORMAT(DATE_ADD(now(), INTERVAL 3 MINUTE), '%Y%m%d%H%i%s');
    set @date = DATE_FORMAT(now(), '%Y%m%d%H%i%s');
    set @mes = p_analyzer_message;
    set @fullmes = p_analyzer_message;
    set p_result_message = '';

    IF @analyzer_type_code in ('SYSMEXSUIT') THEN 
      IF LOCATE('Q|1|-------------------', @mes) > 0 THEN 
        set p_result_code = 'Q';
        set @pf = LOCATE('Q|1|', @mes)+4;
        set @pt = LOCATE('|', @mes, @pf);
        
        set p_barcode = SUBSTRING(@mes, @pf, @pt-@pf);
        set p_result_message = CONCAT('Barcode = ', p_barcode); 
      ELSEIF LOCATE('BX|1|NM|', @mes) > 0 THEN 
        set p_result_code = 'R';
        set p_result_message = '';
        set @pf = LOCATE('BR|1||', @mes)+6;
        set @pt = LOCATE('|', @mes, @pf);
        set p_barcode = TRIM(SUBSTRING(@mes, @pf, @pt-@pf));
        set @i = 1;
        set @pf = LOCATE(CONCAT('BX|',@i,'|NM|'), @mes)+LENGTH(CONCAT('BX|',@i,'|NM|'));
        WHILE @pf > LENGTH(CONCAT('BX|',@i,'|NM|')) AND @i < 50 DO 
          set @pt = LOCATE('|', @mes, @pf);
          set @code = SUBSTRING(@mes, @pf, @pt-@pf); 
          set @code = TRIM(REPLACE(@code, '^',''));
          set @pt = LOCATE('||', @mes, @pf);
          set @pf = @pt+2;
          set @pt = LOCATE('^^', @mes, @pf);
          set @pt2 = LOCATE('|', @mes, @pf);
          IF @pt2 > @pt THEN
            set @value = SUBSTRING(@mes, @pf, @pt-@pf);
          ELSE
            set @value = SUBSTRING(@mes, @pf, @pt2-@pf);
          END IF;
          IF LENGTH(p_barcode) > 5 AND LENGTH(@code) > 0 THEN
            CALL set_lab_attribute(p_barcode, @code, @value, '', @p_res_code, @p_res_message);
            set p_result_message = CONCAT(p_result_message, '\n ', IFNULL(@p_res_message,'')); 
          END IF;
          set @i = @i+1;
          set @pf = LOCATE(CONCAT('BX|',@i,'|NM|'), @mes)+LENGTH(CONCAT('BX|',@i,'|NM|'));
        END WHILE;

      ELSE
        set p_result_code = '';
      END IF;

    ELSEIF @analyzer_type_code = 'INTEGRA' THEN 
      set f = CONCAT('42 ');
      IF LOCATE(f, p_analyzer_message) > 0 THEN 
        set p_result_code = 'Q';
        set @pf = LOCATE(f, p_analyzer_message)+12;
        set p_barcode = SUBSTRING(p_analyzer_message, @pf,14);
        set p_result_message = CONCAT('Barcode = ', p_barcode); 
      ELSEIF LOCATE('53 ', @mes) > 0 THEN 
        set p_result_code = 'R';
        SET @i = 2;
        set @pf = LOCATE('53 ', @mes)+3;
        set p_barcode = SUBSTRING(@mes, @pf,14);

        WHILE LOCATE('55 ', @mes) > 0 DO
            set @pf = LOCATE('55 ', @mes)+3;
            set @code = TRIM( SUBSTRING(@mes, @pf,3));
            set @mes = SUBSTRING(@mes, @pf,3000);
    
            set @pf = LOCATE('00 ', @mes)+3;
            set @value = TRIM(SUBSTRING(@mes, @pf,120));
            set @mes = TRIM(SUBSTRING(@mes, @pf,3000));
    
            set @pt = LOCATE(' ', @mes);
            set @value = TRIM(SUBSTRING(@value, 1,@pt));
            set @value = CAST(@value AS decimal(15,6));

            set @mes = TRIM(SUBSTRING(@mes, @pt,3000));
    
            CALL set_lab_attribute(p_barcode, @code, @value, '', @p_res_code, @p_res_message);
            set p_result_message = CONCAT(p_result_message, '\n ', @p_res_message); 
        END WHILE;

--        set p_result_message = CONCAT('Barcode = ', p_barcode, '/', @code, '/', @value ); 
/*        WHILE @fullmes <> '' DO
          SET @p = 0;
          SET @p = LOCATE(CONCAT('P|',@i,'|||||||||||||||||||||||||||||||||'), @fullmes);

*/
      ELSE
        set p_result_code = '';
      END IF;
--      set p_result_code = '@';
    ELSEIF @analyzer_type_code = 'BIOKSEL' THEN 

      IF LOCATE('Q|1|', @mes) > 0 THEN 
        set p_result_code = 'Q';
        set @pf = LOCATE('Q|1|', @mes)+4;
        set @pt = LOCATE('|', @mes, @pf);

      ELSEIF LOCATE('R|1|', @mes) > 0 THEN

==================== PROCEDURE gen_lab_order ====================
CREATE DEFINER=`root`@`localhost` PROCEDURE `gen_lab_order`(
  IN p_service_barcode varchar(128), 
  IN p_analyzer_id mediumint, 
  IN p_mode varchar(64)
  )
    COMMENT 'v.2.1.179'
BEGIN
    DECLARE result_str varchar(256);
    DECLARE result_blb blob;
    DECLARE t_exec_service_id int UNSIGNED DEFAULT 0;
    DECLARE t_patient_id int UNSIGNED DEFAULT 0;
    DECLARE protocol_type varchar(32);
    
    SELECT aat.analyzer_type_code, aa.analyzer_name, ifnull(JSON_UNQUOTE(JSON_EXTRACT(aa.analyzer_config, '$.protocol_type')),'') 
    INTO @analyzer_type_code, @analyzer_name, protocol_type
     FROM ac_analyzer aa JOIN ac_analyzer_type aat ON aa.analyzer_type_id = aat.analyzer_type_id WHERE aa.analyzer_id = p_analyzer_id;
    
    set @date = DATE_FORMAT(DATE_ADD(now(), INTERVAL 3 MINUTE), '%Y%m%d%H%i%s');

    SELECT l.exec_service_id, mes.patient_id 
    INTO t_exec_service_id, t_patient_id 
    FROM med_exec_service_lab l 
    JOIN med_exec_service mes ON l.exec_service_id = mes.exec_service_id
    WHERE l.service_barcode = p_service_barcode ORDER BY l.exec_service_id DESC LIMIT 1;
    
    IF @analyzer_type_code = 'СOBAS 411' OR @analyzer_type_code = 'SYSMEX XS-1000I' THEN 

      SELECT GROUP_CONCAT(DISTINCT CONCAT('^^^', code_str) SEPARATOR '\\'), s.patient_id, DATE_FORMAT(patient_birthday, '%Y%m%d'), CASE WHEN s.gender_id = 1 THEN 'M' ELSE 'F' END
        INTO @order_list, @patient_id, @patient_birthday, @gender  
      FROM (
          SELECT da.attribute_id , MIN(ifnull(sla.norm_code,da.attribute_code)) code_str, mes.patient_id, mp.patient_birthday, mp.gender_id
          FROM med_exec_service_lab esl
          JOIN med_exec_service mes ON esl.exec_service_id = mes.exec_service_id
          JOIN med_patient mp ON mes.patient_id = mp.patient_id
          JOIN med_exec_service_attribute mesa ON mes.exec_service_id = mesa.exec_service_id
          JOIN dct_attribute da ON mesa.attribute_id = da.attribute_id
          JOIN dct_service_attribute sa ON sa.service_id = mes.service_id AND sa.attribute_id = mesa.attribute_id
          LEFT JOIN dct_service_lab_attribute sla ON sla.service_attribute_id = sa.service_attribute_id AND sla.norm_code > ''
          LEFT JOIN dct_service_lab_method dslm ON sla.method_id = dslm.service_lab_method_id
          WHERE esl.service_barcode = p_service_barcode AND dslm.service_lab_method_code IN ('COBAS', 'COBAS411') AND mes.patient_id = t_patient_id AND mes.patient_id = t_patient_id AND mes.state_id IN (1,2,3,52)
          GROUP BY da.attribute_id ) s WHERE code_str > '';

      IF p_mode = 'COBAS1' THEN
        SELECT CONCAT('1H|\\^&|||HIS01|||||', @analyzer_name, '||P|1|',@date,char(23)) result FROM dual WHERE t_exec_service_id > 0
        UNION ALL
        SELECT CONCAT('2P|1||||', @patient_id,'||',IFNULL(@patient_birthday,''), '|', IFNULL(@gender,''), '||||||||||||||||||||||||||', char(23)) result FROM dual WHERE t_exec_service_id > 0
        UNION ALL
        SELECT CONCAT('3O|1|', p_service_barcode,'||',@order_list, '|R||||||A||||||||||||||O|||||', char(23)) result FROM dual WHERE t_exec_service_id > 0
        UNION ALL
        SELECT CONCAT('4L|1|N', char(13), char(3)) result FROM dual WHERE t_exec_service_id > 0;
      else

        set @result_str = CONCAT('1H|\\^&|||HIS01|||||C111||P|1|',@date,char(13),char(10),
                                 'P|1||||', @patient_id,'||',IFNULL(@patient_birthday,''), '|', IFNULL(@gender,''), '||||||||||||||||||||||||||', char(13),char(10),
                                 'O|1|', p_service_barcode,'||',@order_list, '|R||||||A||||||||||||||O|||||', char(13),char(10),
                                 'L|1|N', char(13), char(3));
  
        SELECT @result_str result FROM dual WHERE t_exec_service_id > 0;
      END IF;

    ELSEIF @analyzer_type_code = 'PENTRA' THEN 

      SELECT GROUP_CONCAT(DISTINCT CONCAT('^^^', code_str) SEPARATOR '\\'), s.registry_code, DATE_FORMAT(patient_birthday, '%Y%m%d'), CASE WHEN s.gender_id = 1 THEN 'M' ELSE 'F' END, s.patient_name
        INTO @order_list, @patient_id, @patient_birthday, @gender, @patient_name 
      FROM (
          SELECT da.attribute_id , MIN(ifnull(sla.norm_code,da.attribute_code)) code_str, mp.registry_code, mp.patient_birthday, mp.gender_id, CONCAT(IFNULL(lp.patient_last_name,''), '^', IFNULL(lp.patient_first_name,''))  patient_name
          FROM med_exec_service_lab esl
          JOIN med_exec_service mes ON esl.exec_service_id = mes.exec_service_id
          JOIN med_patient mp ON mes.patient_id = mp.patient_id
          JOIN lng_patient lp ON mes.patient_id = lp.patient_id
          JOIN med_exec_service_attribute mesa ON mes.exec_service_id = mesa.exec_service_id
          JOIN dct_attribute da ON mesa.attribute_id = da.attribute_id
          JOIN dct_service_attribute sa ON sa.service_id = mes.service_id AND sa.attribute_id = mesa.attribute_id
          LEFT JOIN dct_service_lab_attribute sla ON sla.service_attribute_id = sa.service_attribute_id AND sla.norm_code > ''
          LEFT JOIN dct_service_lab_method dslm ON sla.method_id = dslm.service_lab_method_id
          WHERE esl.service_barcode = p_service_barcode AND dslm.service_lab_method_code IN ('Pentra C400') AND mes.patient_id = t_patient_id AND mes.state_id IN (1,2,3,52)
          GROUP BY da.attribute_id ) s WHERE code_str > '';

        SELECT CONCAT('1H|\\^&|||ABX|||||||P|E1394-97|',@date,char(13),char(3)) result FROM dual WHERE t_exec_service_id > 0
        union ALL
        SELECT CONCAT('2P|1||', @patient_id,'||',@patient_name,'||',IFNULL(@patient_birthday,''), '|', IFNULL(@gender,''), '|||||Prescriptor||||||||||||Location', char(13),char(3)) result FROM dual WHERE t_exec_service_id > 0
        union ALL
        SELECT CONCAT('3O|1|', p_service_barcode,'||',@order_list, '|R|',@date,'|||||N||||1', char(13),char(3)) result FROM dual WHERE t_exec_service_id > 0
        union ALL
        SELECT CONCAT('4L|1|N', char(13), char(3)) result FROM dual WHERE t_exec_service_id > 0;
  


    ELSEIF @analyzer_type_code = 'TOSO' THEN 

      SELECT GROUP_CONCAT(DISTINCT CONCAT('^^^', code_str) SEPARATOR '\\'), s.patient_id, DATE_FORMAT(patient_birthday, '%Y%m%d'), CASE WHEN s.gender_id = 1 THEN 'M' ELSE 'F' END
        INTO @order_list, @patient_id, @patient_birthday, @gender  
      FROM (
          SELECT da.attribute_id , MIN(ifnull(sla.norm_code,da.attribute_code)) code_str, mes.patient_id, mp.patient_birthday, mp.gender_id
          FROM med_exec_service_lab esl
          JOIN med_exec_service mes ON esl.exec_service_id = mes.exec_service_id
          JOIN med_patient mp ON mes.patient_id = mp.patient_id

          JOIN med_exec_service_attribute mesa ON mes.exec_service_id = mesa.exec_service_id
          JOIN dct_attribute da ON mesa.attribute_id = da.attribute_id
          JOIN dct_service_attribute sa ON sa.service_id = mes.service_id AND sa.attribute_id = mesa.attribute_id
          LEFT JOIN dct_service_lab_attribute sla ON sla.service_attribute_id = sa.service_attribute_id AND sla.norm_code > ''
          LEFT JOIN dct_service_lab_method dslm ON sla.method_id = dslm.service_lab_method_id
          WHERE esl.service_barcode = p_service_barcode 
          AND mes.patient_id = t_patient_id 
          GROUP BY da.attribute_id 
          ) s WHERE code_str > '';

          set @result_str = CONCAT('1H|\\^&|||MINDRAY240^01.03.07.03^123456|||||||SA|1394-97|',@date, char(13),
          'P|1|||',t_patient_id,'|',@patient_name,'||',@patient_birthday,'|',@gender,'||||||||||||||||||||||||||', char(13),
          @order_list, 'L|1|N', char(13), char(3));
          SELECT @result_str result FROM dual WHERE t_exec_service_id > 0;
    ELSE
      SELECT @result_str result;
    END IF;
END

==================== PROCEDURE gen_lab_tube_barcode ====================
CREATE DEFINER=`root`@`localhost` PROCEDURE `gen_lab_tube_barcode`(IN p_exec_service_id mediumint UNSIGNED,
IN p_visit_id mediumint UNSIGNED)
BEGIN

  DECLARE t_barcode varchar(64);
  DECLARE t_tube_type_id mediumint(9) UNSIGNED;
  DECLARE t_service_id mediumint(9) UNSIGNED;

  DECLARE I integer DEFAULT 0;
  DECLARE RL1 integer DEFAULT 0;
  DECLARE RL2 integer DEFAULT 0;

  DECLARE done int DEFAULT 0;

  SELECT
    mes.service_id INTO t_service_id
  FROM med_exec_service mes
  WHERE mes.exec_service_id = p_exec_service_id;

  IF NOT EXISTS (SELECT
        1
      FROM med_exec_service_lab
      WHERE exec_service_id = p_exec_service_id) THEN
    INSERT INTO med_exec_service_lab (exec_service_id, department_id, manipulation_id, biomaterial_type_id, service_volume, report_id, service_lab_method_id, tube_type_id, partner_id, service_barcode)
      SELECT
        p_exec_service_id,
        dsl.department_id,
        dsl.manipulation_id,
        dsl.biomaterial_type_id,
        dsl.service_volume,
        dsl.report_id,
        dsl.service_lab_method_id,
        dsl.tube_type_id,
        dsl.partner_id,
        NULL
      FROM dct_service_lab dsl
      WHERE dsl.service_id = t_service_id;
  END IF;

  SELECT
    mesl.tube_type_id INTO t_tube_type_id
  FROM med_exec_service_lab mesl
  WHERE mesl.exec_service_id = p_exec_service_id;

  SET @t_counter_value = 0;

  UPDATE dct_counter dc
  SET dc.counter_value = (@t_counter_value := dc.counter_value + 1)
  WHERE dc.counter_code = 'lab_tube_barcode_ean8';

  SET t_barcode = CONCAT(SUBSTRING(CONCAT('00000000', CAST(@t_counter_value AS char)) FROM (CHAR_LENGTH(CONCAT('00000000', CAST(@t_counter_value AS char))) - 6) FOR 7), '0');

  IF (SUBSTRING(t_barcode FROM 1 FOR 1) = '0') THEN
    SET t_barcode = CONCAT('1', SUBSTRING(t_barcode FROM 2 FOR 7));
  END IF;

  SET I = 8;
  WHILE (I > 0) DO
    IF (MOD(I, 2) = 0) THEN
      SET RL2 = RL2 + CAST(SUBSTRING(t_barcode FROM I FOR 1) AS decimal) * 1;
    ELSE
      SET RL2 = RL2 + CAST(SUBSTRING(t_barcode FROM I FOR 1) AS decimal) * 3;
    END IF;

    SET I = I - 1;
  END WHILE;

  SET RL1 = MOD(RL2, 10);
  IF (RL1 <> 0) THEN
    SET RL2 = 10 - RL1;
  ELSE
    SET RL2 = 0;
  END IF;

  SET t_barcode = CONCAT(SUBSTRING(t_barcode FROM 1 FOR 7), CAST(RL2 AS char));

  UPDATE med_exec_service_lab mesl
  SET mesl.service_barcode = t_barcode
  WHERE mesl.exec_service_id = p_exec_service_id
  AND mesl.service_barcode IS NULL;
END

==================== PROCEDURE list_exec_service_lab ====================
CREATE DEFINER=`root`@`localhost` PROCEDURE `list_exec_service_lab`(IN p_id mediumint UNSIGNED)
    COMMENT 'v.2.0.246'
BEGIN

  SELECT
    dsl.exec_service_id,
    dsl.department_id,
    dd.department_name,
    dsl.tube_type_id,
    dltt.tube_type_name,
    dsl.manipulation_id,
    dsm.service_name AS manipulation_name,
    dsl.biomaterial_type_id,
    dlbt.biomaterial_type_name,
    dsl.service_volume,
    dsl.report_id,
    rr.report_name,
    dsl.service_lab_method_id,
    dslm.service_lab_method_name,
    dsl.partner_id,
    dp.partner_name,
    dsl.service_barcode
  FROM med_exec_service_lab dsl
    LEFT OUTER JOIN dct_department dd
      ON dd.department_id = dsl.department_id
    LEFT OUTER JOIN dct_partner dp
      ON dp.partner_id = dsl.partner_id
    LEFT OUTER JOIN dct_service dsm
      ON dsm.service_id = dsl.manipulation_id
    LEFT OUTER JOIN dct_lab_biomaterial_type dlbt
      ON dlbt.biomaterial_type_id = dsl.biomaterial_type_id
    LEFT OUTER JOIN dct_service_lab_method dslm
      ON dslm.service_lab_method_id = dsl.service_lab_method_id
    LEFT OUTER JOIN rep_report rr
      ON rr.report_id = dsl.report_id
    LEFT OUTER JOIN dct_lab_tube_type dltt
      ON dltt.tube_type_id = dsl.tube_type_id
  WHERE dsl.exec_service_id = p_id;
END
